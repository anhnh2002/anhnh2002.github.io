\begin{rSection}{Experience}

\textbf{FPT Software AI Center} \hfill May 2025 -- Present\\
\textit{AI Research Resident} \hfill \textit{Hanoi, Vietnam}\\
Advisors: Associate Prof.\ Bach Le, University of Melbourne;
Assistant Prof.\ Nghi D. Q. Bui, VinUniversity

\begin{itemize}[leftmargin=1.5em,itemsep=0em,topsep=0.1em]
    \item \textbf{CodeWiki} (first author, Findings of ACL 2026): built an agentic
    framework for repository-level documentation across seven languages by
    decomposing the dependency graph into a module tree, delegating leaf modules
    to documentation agents, and synthesizing parent-level text, architecture and
    data-flow diagrams bottom-up
    \item Built \textbf{CodeWikiBench}, a benchmark with hierarchical rubrics and
    agentic LLM-judge evaluation; CodeWiki reaches a 68.79\% quality score,
    +4.73 points over closed-source DeepWiki and +21.66 points over OpenDeepWiki,
    with gains of up to 18.54\% on individual repositories
    \item \textbf{RustPrint}: co-developed agentic C-to-Rust migration that uses
    CodeWiki documentation as a blueprint and documentation mismatches as repair
    signals; on eight C repositories (11K--84K LoC) it compiles every target where
    prior LLM translators fail and, with Kimi-K2, beats Claude Code on feature
    preservation (93.26\% vs.\ 52.52\%) and test pass rate (95.17\% vs.\ 79.85\%)
    \item \textbf{FALCON} (under review): co-developed a security verifier that
    separates attack exploration (proof-of-concept probes) from judging their
    execution traces; as repair feedback it raises CWEval Func-Sec@1 by 25.2
    points where LLM-as-a-judge brings almost no gain, and as an RL reward it
    beats the REAL and SecCodePRM rewards
    \item \textbf{SecPRBench} (under review, with Prof.\ Lingming Zhang, UIUC):
    curated the FFmpeg, FreeType, and SQLite tasks of a benchmark of 265
    vulnerability-introducing changes from nine C, C++, and Rust projects, each
    paired with its repaired version; agents on eight model backends detect only
    32.5\%--78.1\% of the target vulnerabilities
\end{itemize}

\textbf{Research Assistant} \hfill March 2024 -- April 2025\\
\textit{Continual Learning and Natural Language Processing Research}
\hfill \textit{Hanoi, Vietnam}\\
Advisors: Assistant Prof.\ Linh Ngo Van; Associate Prof.\ Thien Huu Nguyen

\begin{itemize}[leftmargin=1.5em,itemsep=0em,topsep=0.1em]
    \item \textbf{Mutual-pairing data augmentation} (co-first author, NAACL 2025):
    combined old and new relation samples into composite inputs with objectives
    that induce adversarial-training effects, cutting the accuracy drop to 26.40\%
    on FewRel and 24.36\% on TACRED (2.49\% better than CPL on TACRED); showed
    that Sharpness-Aware Minimization costs over 6\% accuracy with LLM2Vec
    \item \textbf{Generalization preservation} (co-first author, EMNLP 2024):
    reused discarded pre-trained LM heads through mutual-information
    maximization with the classifier, improving final accuracy by about 2\% for
    CPL and ConPL, 1\% for SCKD, and up to 3.75\% with LLaMA-2-7B
\end{itemize}
